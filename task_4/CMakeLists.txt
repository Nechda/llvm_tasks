cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(task_4 LANGUAGES C CXX)
else()
  enable_language(CXX)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(LLVM REQUIRED CONFIG)
message(STATUS "Using LLVM ${LLVM_PACKAGE_VERSION} from ${LLVM_DIR}")
add_executable(asm2ir main.cpp asm.cpp ir.cpp runtime.cpp engine.cpp)
target_compile_options(asm2ir PRIVATE -ffp-contract=off)
add_executable(asm_sim sim_main.cpp asm.cpp simulator.cpp runtime.cpp)
target_compile_options(asm_sim PRIVATE -ffp-contract=off)
target_include_directories(asm2ir SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
separate_arguments(TASK4_LLVM_DEFINITIONS NATIVE_COMMAND "${LLVM_DEFINITIONS}")
target_compile_options(asm2ir PRIVATE ${TASK4_LLVM_DEFINITIONS})
llvm_map_components_to_libnames(TASK4_LLVM_LIBS Core Support Passes ExecutionEngine MCJIT native)
target_link_libraries(asm2ir PRIVATE ${TASK4_LLVM_LIBS})

option(TASK4_USE_SDL "Link task_4 with real llvm_course SDL functions" OFF)
if(TASK4_USE_SDL)
  set(LLVM_COURSE_ROOT "${LLVM_COURSE_ROOT}" CACHE PATH "Path to llvm_course")
  if(NOT EXISTS "${LLVM_COURSE_ROOT}/SDL/sim.c")
    message(FATAL_ERROR "TASK4_USE_SDL requires LLVM_COURSE_ROOT with SDL/sim.c")
  endif()
  find_package(SDL2 REQUIRED)
  foreach(target asm2ir asm_sim)
    target_sources(${target} PRIVATE "${LLVM_COURSE_ROOT}/SDL/sim.c")
    if(TARGET SDL2::SDL2)
      target_link_libraries(${target} PRIVATE SDL2::SDL2)
    else()
      target_include_directories(${target} PRIVATE ${SDL2_INCLUDE_DIRS})
      target_link_libraries(${target} PRIVATE ${SDL2_LIBRARIES})
    endif()
  endforeach()
else()
  target_sources(asm2ir PRIVATE ../task_3/sim_stub.cpp)
  target_sources(asm_sim PRIVATE ../task_3/sim_stub.cpp)
endif()

find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND AND NOT TASK4_USE_SDL)
  enable_testing()
  add_test(NAME task4_isa COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/test_isa.py --exe $<TARGET_FILE:asm2ir>
    --sim $<TARGET_FILE:asm_sim>)
  add_custom_target(task4_check
    COMMAND ${CMAKE_CTEST_COMMAND} --test-dir ${CMAKE_CURRENT_BINARY_DIR} --output-on-failure
    DEPENDS asm2ir asm_sim USES_TERMINAL)
endif()
find_program(TASK4_CLANG NAMES clang HINTS "${LLVM_TOOLS_BINARY_DIR}")
if(Python3_Interpreter_FOUND AND TASK4_CLANG AND NOT TASK4_USE_SDL)
  add_test(NAME task4_frames COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/check.py --exe $<TARGET_FILE:asm2ir>
    --sim $<TARGET_FILE:asm_sim> --clang ${TASK4_CLANG} --cxx ${CMAKE_CXX_COMPILER}
    "--cxx-flags=${CMAKE_CXX_FLAGS}" "--link-flags=${CMAKE_EXE_LINKER_FLAGS}")
endif()
